# PHASE 8, step G3: the app's native engine (app/src/main/cpp/llama_engine.cpp) and its vendored llama.cpp, built for the
# CI computer (Linux x86-64) and run against a tiny real GGUF model. Same sources as the phone's library; the only extra
# is llama.cpp's x86 CPU code (ggml-cpu/arch/x86), which the app does not vendor: run.sh copies it from the same commit.
cmake_minimum_required(VERSION 3.18)
project(sphynx_native_test C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CPP ${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp)
set(LLAMA_DIR ${CPP}/llama.cpp)

# the same settings as app/src/main/cpp/CMakeLists.txt, read from it so the pin can't drift
file(READ ${CPP}/CMakeLists.txt APP_CMAKE)
string(REGEX MATCH "set\\(LLAMA_PIN_TAG +\"([^\"]+)\"\\)" _ "${APP_CMAKE}")
set(LLAMA_PIN_TAG ${CMAKE_MATCH_1})
string(REGEX MATCH "set\\(LLAMA_PIN_COMMIT +\"([^\"]+)\"\\)" _ "${APP_CMAKE}")
set(LLAMA_PIN_COMMIT ${CMAKE_MATCH_1})
string(REGEX MATCH "b([0-9]+)" _ "${LLAMA_PIN_TAG}")
set(LLAMA_BUILD_NUMBER ${CMAKE_MATCH_1})
set(LLAMA_VERSION_MAJOR 0)
set(LLAMA_VERSION_MINOR 7)
set(LLAMA_VERSION_PATCH 0)
set(LLAMA_VERSION_BASE "${LLAMA_VERSION_MAJOR}.${LLAMA_VERSION_MINOR}.${LLAMA_VERSION_PATCH}")
set(LLAMA_VERSION "${LLAMA_VERSION_BASE}-${LLAMA_PIN_TAG}")
set(LLAMA_BUILD_COMMIT ${LLAMA_PIN_COMMIT})
set(GGML_BUILD_NUMBER  ${LLAMA_BUILD_NUMBER})
set(GGML_BUILD_COMMIT  ${LLAMA_PIN_COMMIT})
function(llama_add_compile_flags)
endfunction()

set(CMAKE_BUILD_TYPE Release)
set(BUILD_SHARED_LIBS     OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL       OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE           OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP           OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE        ON  CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS      OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES   OFF CACHE BOOL "" FORCE)

add_subdirectory(${LLAMA_DIR}/ggml ggml)
add_subdirectory(${LLAMA_DIR}/src  llama)
set(LLAMA_BUILD_TOOLS OFF)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_DIR}/vendor/hash      vendor-hash)
add_subdirectory(${LLAMA_DIR}/vendor/miniaudio vendor-miniaudio)
add_subdirectory(${LLAMA_DIR}/vendor/stb       vendor-stb)
add_subdirectory(${LLAMA_DIR}/vendor/sheredom  vendor-sheredom)
add_subdirectory(${LLAMA_DIR}/tools/mtmd       mtmd)

add_executable(engine_test engine_test.cpp ${CPP}/llama_engine.cpp)
target_include_directories(engine_test PRIVATE ${CPP})
target_compile_definitions(engine_test PRIVATE LLAMA_PIN_TAG="${LLAMA_PIN_TAG}" LLAMA_PIN_COMMIT="${LLAMA_PIN_COMMIT}")
target_link_libraries(engine_test PRIVATE mtmd llama ggml)
